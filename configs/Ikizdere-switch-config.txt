! ============================================================
! Switch: Ikizdere (Rize Region — 10.53.2.0/24)
! Role: Ikizdere District Access Switch
! Management IP: 10.53.2.254/24 | Gateway: 10.53.2.1
! ============================================================

enable
configure terminal

hostname Ikizdere

! --- VLAN 1 Management Interface ---
interface vlan 1
 no shutdown
 ip address 10.53.2.254 255.255.255.0
exit

! --- Default Gateway ---
ip default-gateway 10.53.2.1

! --- Disable DNS lookup ---
no ip domain lookup

! --- Enable password ---
enable secret cisco

! --- SSH Setup ---
ip domain-name cisco
service password-encryption
crypto key generate rsa
! (enter 1024 when prompted)

! --- Local Users ---
username admin privilege 15 secret admin
username user privilege 1 secret user

! --- VTY Lines (Telnet + SSH) ---
line vty 0 4
 logging synchronous
 login local
 transport input all
exit

! --- SSH Settings ---
ip ssh version 2
ip ssh authentication-retries 4
ip ssh time-out 30

! --- Console Line ---
line console 0
 logging synchronous
 login local
exit

end
